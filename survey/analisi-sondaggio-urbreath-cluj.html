6. Schema DB — 4 tabelle flat consigliate

Seguendo il pattern dei file DML esistenti nel progetto (es. bi_registry/, bi_materialized_views/), si propone di creare 4 tabelle aggregated-flat con INSERT statici one-shot dall'Excel.

-- ── Tabella 1: Problemi per sito ─────────────────────────────
CREATE TABLE survey_site_problem (
    site_id           VARCHAR(20)     NOT NULL,   -- 'Site1', 'Site2', 'Site3', 'Site4'
    problem_label     VARCHAR(200)    NOT NULL,
    mention_count     INTEGER         NOT NULL,
    respondents_total INTEGER         NOT NULL,
    pct_respondents   NUMERIC(5,2)    NOT NULL    -- mention_count / respondents_total * 100
);

-- ── Tabella 2: Soddisfazione media per sito e indicatore ──────
CREATE TABLE survey_satisfaction_avg (
    site_id           VARCHAR(20)     NOT NULL,
    indicator_label   VARCHAR(200)    NOT NULL,
    avg_score         NUMERIC(4,2),                  -- NULL se sito non ha dati
    respondents_total INTEGER         NOT NULL
);

-- ── Tabella 3: Consapevolezza NBS per fascia d'età ────────────
CREATE TABLE survey_nbs_awareness (
    age_group         VARCHAR(20)     NOT NULL,   -- '16-24', '25-34', ..., '65+'
    familiarity_level VARCHAR(60)     NOT NULL,   -- 4 livelli ordinati
    familiarity_rank  INTEGER         NOT NULL,   -- 1..4 per ordinamento stacked bar
    count             INTEGER         NOT NULL,
    pct_age_group     NUMERIC(5,2)    NOT NULL    -- count / tot_per_age * 100
);

-- ── Tabella 4: Cambiamenti desiderati per sito ────────────────
CREATE TABLE survey_desired_change (
    site_id           VARCHAR(20)     NOT NULL,
    change_label      VARCHAR(200)    NOT NULL,
    mention_count     INTEGER         NOT NULL,
    respondents_total INTEGER         NOT NULL,
    pct_respondents   NUMERIC(5,2)    NOT NULL,
    rank_in_site      INTEGER         NOT NULL    -- per filtrare top-5 per sito
);
